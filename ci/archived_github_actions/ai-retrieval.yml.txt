# Generated retrieval workflow; engine 1.1.0. Generic code only; no configured secrets.
name: Refresh repository retrieval

on:
  push:
  workflow_dispatch:
  workflow_call:
  schedule:
    - cron: '23 */6 * * *'

permissions:
  contents: write

concurrency:
  group: ai-retrieval-${{ github.repository }}-${{ github.ref }}
  cancel-in-progress: false

jobs:
  retrieval:
    if: github.ref_type == 'branch'
    runs-on: ubuntu-latest
    timeout-minutes: 20
    env:
      ENGINE_URL: https://raw.githubusercontent.com/Otdel1C/PerformanceMeasurementsAnalyzer-for-1C-Enterprise-8.x/bbe1e54374b1a4f62aa341374fe08d7974a1e8f8/.github/retrieval_index.py
      ENGINE_SHA256: e8973f5bd746412df17074ea758b88deb654eaabfed374be38a799cd64dacbd1
      GIT_LFS_SKIP_SMUDGE: '1'
    steps:
      - name: Fetch selected branch and bounded ownership history
        shell: bash
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          set -euo pipefail
          git init -q .
          git remote add origin "https://github.com/$GITHUB_REPOSITORY.git"
          auth="AUTHORIZATION: basic $(printf 'x-access-token:%s' "$GH_TOKEN" | base64 -w0)"
          git -c http.https://github.com/.extraheader="$auth" fetch --depth=64 origin "refs/heads/$GITHUB_REF_NAME"
          git -c core.hooksPath=/dev/null checkout --detach FETCH_HEAD
          git config user.name 'github-actions[bot]'
          git config user.email '41898282+github-actions[bot]@users.noreply.github.com'

      - name: Verify pinned readable engine and build with controlled recovery
        shell: bash
        run: |
          set -euo pipefail
          python3 -I - <<'PY_ENGINE'
          import hashlib, os, pathlib, urllib.request
          with urllib.request.urlopen(os.environ['ENGINE_URL'], timeout=60) as response:
              engine = response.read(1048577)
          if len(engine) > 1048576 or hashlib.sha256(engine).hexdigest() != os.environ['ENGINE_SHA256']:
              raise SystemExit('Pinned engine checksum mismatch')
          pathlib.Path(os.environ['RUNNER_TEMP'], 'retrieval_index.py').write_bytes(engine)
          PY_ENGINE
          python3 -I "$RUNNER_TEMP/retrieval_index.py" --repair
          python3 -I "$RUNNER_TEMP/retrieval_index.py" --check

      - name: Publish verified index with an immutable manifest proof
        shell: bash
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          set -euo pipefail
          git add -f -- .ai/INDEX.md .ai/retrieval
          python3 -I - <<'PY_SCOPE'
          import re, subprocess
          paths = subprocess.check_output(['git', 'diff', '--cached', '--name-only', '-z']).split(b'\0')
          owned = re.compile(rb'^\.ai/retrieval/[0-9a-f]{16}/(?:ROUTE\.md|(?:files|symbols|headings)-[0-9]{4,}\.jsonl)$')
          for path in filter(None, paths):
              if path not in (b'.ai/INDEX.md', b'.ai/retrieval/manifest.json') and not owned.fullmatch(path):
                  raise SystemExit('Refusing to publish an unowned path')
          PY_SCOPE
          source=$(git rev-parse HEAD)
          auth="AUTHORIZATION: basic $(printf 'x-access-token:%s' "$GH_TOKEN" | base64 -w0)"
          current=$(git -c http.https://github.com/.extraheader="$auth" ls-remote origin "refs/heads/$GITHUB_REF_NAME" | cut -f1)
          if [ "$source" != "$current" ]; then
            echo 'Branch advanced; no stale index was pushed. Verify the newer run or dispatch a rebuild.'
            exit 1
          fi
          if ! git diff --cached --quiet; then
            git -c core.hooksPath=/dev/null commit -m 'chore(ai): refresh verified retrieval index'
            git -c http.https://github.com/.extraheader="$auth" push origin "HEAD:refs/heads/$GITHUB_REF_NAME"
          fi
          python3 -I "$RUNNER_TEMP/retrieval_index.py" --check
          printf 'RETRIEVAL_PROOF commit=%s manifest_blob=%s engine_sha256=e8973f5bd746412df17074ea758b88deb654eaabfed374be38a799cd64dacbd1\n' \
            "$(git rev-parse HEAD)" "$(git rev-parse HEAD:.ai/retrieval/manifest.json)"
